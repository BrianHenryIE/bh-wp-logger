2/**
3 * Strips `WP_Hook` objects from logged backtraces.
4 *
5 * Backtraces added to the log context can contain `WP_Hook` objects, whose `callbacks` property lists every
6 * callback every plugin has registered on that hook, which makes the log file size explode. They appear in
7 * two places:
8 *
9 * * `context.debug_backtrace[].object` — `debug_backtrace()` includes each frame's live object.
10 * * `context.exception.backtrace[].args` — `Throwable::getTrace()` frames omit `object` but include `args`,
11 *   which can carry a `WP_Hook` (e.g. an exception thrown inside a hook callback).
12 *
13 * This Monolog processor replaces each `WP_Hook` found in either with its class name before the record is written.
14 *
15 * Monolog's generic lever for oversized context is `NormalizerFormatter`'s `maxNormalizeDepth` /
16 * `maxNormalizeItemCount`, but those truncate indiscriminately; this processor targets the known
17 * offender without losing other context.
18 *
19 * @see \BrianHenryIE\WP_Logger\API\API::get_backtrace()
20 * @see \BrianHenryIE\WP_Logger\API\BH_WP_PSR_Logger::log()
21 * @see \Monolog\Formatter\NormalizerFormatter
22 *
23 * @package brianhenryie/bh-wp-logger
24 */
25
26namespace BrianHenryIE\WP_Logger\API;
27
28use Monolog\LogRecord;
29use Monolog\Processor\ProcessorInterface;
30use WP_Hook;

32/**
33 * Monolog processor which replaces `WP_Hook` objects in the record's backtrace context with their class name.
34 *
35 * NB: Backtrace frames reference the live `WP_Hook` objects in `$wp_filter`, so they must never be modified —
36 * e.g. emptying `WP_Hook::$callbacks` would deregister every callback on that hook for the remainder of the
37 * request. The frame entry is instead replaced in a copy of the context array.
38 */

43     *
44     * Checks both the `debug_backtrace` context and the backtrace recorded from a logged exception.
45     *
46     * @param LogRecord $record The Monolog log record, whose context may contain backtraces.
47     *
48     * @return LogRecord The record, with a modified copy of the context when a `WP_Hook` was present.
49     */

56            $context['debug_backtrace'] = $this->remove_wp_hooks_from_frames( $context['debug_backtrace'], $changed );

59        if ( isset( $context['exception']['backtrace'] ) && is_array( $context['exception']['backtrace'] ) ) {

61        }

64    }

66    /**
67     * Replace `WP_Hook` objects found in each frame's `object` or among its `args` with the class name string.
68     *
69     * Only those two locations are checked — deeper structures inside `args` are intentionally not walked,
70     * since a `WP_Hook` nested further than a direct argument has not been observed in practice.

74     *

76     */
77    protected function remove_wp_hooks_from_frames( array $frames, bool &$changed ): array {
78
79        foreach ( $frames as $frame_index => $frame ) {
80            if ( ! is_array( $frame ) ) {
81                continue;
82            }
83
84            if ( isset( $frame['object'] ) && $frame['object'] instanceof WP_Hook ) {
85                $frames[ $frame_index ]['object'] = WP_Hook::class;
86                $changed                          = true;
87            }
88
89            if ( isset( $frame['args'] ) && is_array( $frame['args'] ) ) {

93
94                        $changed = true;
95                    }
96                }
97            }
98        }
